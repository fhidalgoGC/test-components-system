import { useState, useCallback, type ReactNode } from "react";
import { useIsMobile } from "GC-UI-COMPONENTS/client/src/lib/ui-library/hooks";
import type { LayoutColumnComponent } from "GC-UI-COMPONENTS/client/src/lib/ui-library/components/LayoutColumn";
import {
  LayoutColumn,
} from "GC-UI-COMPONENTS/client/src/lib/ui-library/components/LayoutColumn";
import { CardWrapper } from "@/components/general/wrappers";
import { useDataListFilters } from "../hooks";
import {
  HeaderDataList,
  FiltersDataList,
  ViewDataList,
  ViewDataListCards,
  PaginatorDataList,
} from "../components";
import { OptionsFiltersDataList } from "../components/optionsFiltersDataList";
import type { DataListFiltersProps, CurrentView, GroupByOption, GroupByOptionItem } from "../types";

const VIEW_LIMITS = { table: 5, grid: 12, group: 5 };

interface MobileDataListFiltersProps {
  className?: string;
  headerContent?: ReactNode;
  onToggleFilterSection: () => void;
  onViewChange: (view: "list" | "grid") => void;
  activeView: "list" | "grid";
  onDateChange: (date: Date | undefined) => void;
  state: Record<string, unknown>;
  groupBy: GroupByOption | null;
  onGroupByChange: (newGroupBy: GroupByOption | null) => void;
  onSearchChange: (value: string) => void;
  groupByOptions?: GroupByOptionItem[];
  filterContent?: DataListFiltersProps["filterContent"];
  currentView: CurrentView;
  mobileViewContent: ReactNode;
  mobilePaginatorOptions: number[];
  mobileDefaultLimit: number;
  paginatorRightContent?: ReactNode;
}

function MobileDataListFilters({
  className,
  headerContent,
  onToggleFilterSection,
  onViewChange,
  activeView,
  onDateChange,
  state,
  groupBy,
  onGroupByChange,
  onSearchChange,
  groupByOptions,
  filterContent,
  currentView,
  mobileViewContent,
  mobilePaginatorOptions,
  mobileDefaultLimit,
  paginatorRightContent,
}: MobileDataListFiltersProps) {
  const [mobileFilterVisible, setMobileFilterVisible] = useState(false);

  const handleToggleFilter = useCallback(() => {
    setMobileFilterVisible((prev) => !prev);
    onToggleFilterSection();
  }, [onToggleFilterSection]);

  const handleFilterClose = useCallback(() => {
    setMobileFilterVisible(false);
  }, []);

  return (
    <div className={`flex flex-col w-full bg-white rounded-xl border border-gray-200 overflow-hidden ${className ?? ""}`} data-testid="data-list-filters-mobile">
      <HeaderDataList>{headerContent}</HeaderDataList>
      <div className="w-full h-px bg-gray-200" />
      <OptionsFiltersDataList
        onToggleFilter={handleToggleFilter}
        isFilterVisible={mobileFilterVisible}
        onViewChange={onViewChange}
        activeView={activeView}
        onDateChange={onDateChange}
        initialDate={state.start_time as string | undefined}
        groupBy={groupBy}
        onGroupByChange={onGroupByChange}
        searchValue={(state.textSearch as string) ?? ""}
        onSearchChange={onSearchChange}
        groupByOptions={groupByOptions}
      />
      {mobileFilterVisible && (
        <>
          <div className="w-full h-px bg-gray-200" />
          <FiltersDataList>
            {typeof filterContent === "function"
              ? filterContent(activeView, currentView, handleFilterClose)
              : filterContent}
          </FiltersDataList>
        </>
      )}
      <div className="w-full h-px bg-gray-200" />
      {mobileViewContent}
      <div className="w-full h-px bg-gray-200" />
      <PaginatorDataList itemsPerPageOptions={mobilePaginatorOptions} defaultItemsPerPage={mobileDefaultLimit} rightContent={paginatorRightContent} showItemsPerPage={false} />
    </div>
  );
}

export function DataListFilters({ className, renderCardItem, columns, tableConfig, filterContent, headerContent, paginatorRightContent, groupBodyComponent, groupReloadOnOpen, groupByOptions }: DataListFiltersProps) {
  const isMobile = useIsMobile();
  
  const initialComponents: LayoutColumnComponent[] = [
    {
      id: "filter-section-1",
      component: (
        <div className="w-full h-full flex flex-col">
          <div className="flex-1 min-h-0">
            <HeaderDataList>{headerContent}</HeaderDataList>
          </div>
          <div className="w-full h-px bg-gray-200" data-testid="divider-section-one" />
        </div>
      ),
      align: "top",
      slot: 0,
      sizeMode: "full",
    },
    {
      id: "filter-section-2",
      component: null,
      align: "top",
      slot: 1,
      sizeMode: "full",
    },
    {
      id: "filter-section-3",
      component: null,
      align: "top",
      slot: 2,
      sizeMode: "full",
    },
    {
      id: "filter-section-4",
      component: (
        <div className="w-full h-full flex flex-col">
          <div className="flex-1 min-h-0">
            <ViewDataList columns={columns ?? []} tableConfig={tableConfig ?? { columns: [] }} />
          </div>
          <div className="w-full h-px bg-gray-200" data-testid="divider-section-four" />
        </div>
      ),
      align: "top",
      slot: 3,
      sizeMode: "full",
    },
    {
      id: "filter-section-5",
      component: <PaginatorDataList itemsPerPageOptions={[5, 10, 20]} defaultItemsPerPage={VIEW_LIMITS.table} rightContent={paginatorRightContent} />,
      align: "top",
      slot: 4,
      sizeMode: "full",
    },
  ];

  const {
    slotConfig,
    currentView,
    activeView,
    groupBy,
    state,
    isFilterVisible,
    controller,
    onToggleFilterSection,
    onSearchChange,
    onDateChange,
    onViewChange,
    onGroupByChange,
  } = useDataListFilters({
    columns,
    tableConfig,
    renderCardItem,
    paginatorRightContent,
    groupBodyComponent,
    groupReloadOnOpen,
    initialComponents,
  });

  if (isMobile) {
    const mobileViewContent = currentView === "grid" && renderCardItem
      ? <ViewDataListCards renderItem={renderCardItem} />
      : <ViewDataList columns={columns ?? []} tableConfig={tableConfig ?? { columns: [] }} />;

    const mobilePaginatorOptions = currentView === "grid" ? [4, 8, 12] : [5, 10, 20];
    const mobileDefaultLimit = VIEW_LIMITS[currentView] ?? VIEW_LIMITS.table;

    return (
      <MobileDataListFilters
        className={className}
        headerContent={headerContent}
        onToggleFilterSection={onToggleFilterSection}
        onViewChange={onViewChange}
        activeView={activeView}
        onDateChange={onDateChange}
        state={state}
        groupBy={groupBy}
        onGroupByChange={onGroupByChange}
        onSearchChange={onSearchChange}
        groupByOptions={groupByOptions}
        filterContent={filterContent}
        currentView={currentView}
        mobileViewContent={mobileViewContent}
        mobilePaginatorOptions={mobilePaginatorOptions}
        mobileDefaultLimit={mobileDefaultLimit}
        paginatorRightContent={paginatorRightContent}
      />
    );
  }

  const updatedComponents: LayoutColumnComponent[] = [
    initialComponents[0],
    {
      ...initialComponents[1],
      component: (
        <div className="w-full h-full flex flex-col">
          <div className="flex-1 min-h-0">
            <OptionsFiltersDataList
              onToggleFilter={onToggleFilterSection}
              isFilterVisible={isFilterVisible}
              onViewChange={onViewChange}
              activeView={activeView}
              onDateChange={onDateChange}
              initialDate={state.start_time as string | undefined}
              groupBy={groupBy}
              onGroupByChange={onGroupByChange}
              searchValue={(state.textSearch as string) ?? ""}
              onSearchChange={onSearchChange}
              groupByOptions={groupByOptions}
            />
          </div>
          <div className="w-full h-px bg-gray-200" data-testid="divider-section-two" />
        </div>
      ),
    },
    {
      ...initialComponents[2],
      component: (
        <div className="w-full h-full flex flex-col">
          <div className="flex-1 min-h-0">
            <FiltersDataList>{typeof filterContent === "function" ? filterContent(activeView, currentView) : filterContent}</FiltersDataList>
          </div>
          <div className="w-full h-px bg-gray-200" data-testid="divider-section-three" />
        </div>
      ),
    },
    initialComponents[3],
    initialComponents[4],
  ];

  return (
    <CardWrapper
      padding="none"
      className={`overflow-hidden ${className ?? ""}`}
      dataTestId="card-data-list-filters"
    >
      <LayoutColumn
        slots={5}
        slotConfig={slotConfig}
        controller={controller}
        widthMode="full"
        heightMode="auto"
        slotGap="none"
        components={updatedComponents}
      />
    </CardWrapper>
  );
}

import { useEffect } from "react";
import { X } from "lucide-react";

type HeightMode = "auto" | "full" | "half";

const heightClasses: Record<HeightMode, string> = {
  auto: "max-h-[85vh]",
  full: "h-[95vh]",
  half: "h-[50vh]",
};

interface BottomSheetWrapperProps {
  isOpen: boolean;
  onClose: () => void;
  title?: string;
  subtitle?: string;
  header?: React.ReactNode;
  children: React.ReactNode;
  footer?: React.ReactNode;
  showCloseButton?: boolean;
  heightMode?: HeightMode;
  className?: string;
  dataTestId?: string;
}

export function BottomSheetWrapper({
  isOpen,
  onClose,
  title,
  subtitle,
  header,
  children,
  footer,
  showCloseButton = true,
  heightMode = "auto",
  className = "",
  dataTestId = "bottom-sheet",
}: BottomSheetWrapperProps) {
  useEffect(() => {
    if (isOpen) {
      document.body.style.overflow = "hidden";
    } else {
      document.body.style.overflow = "";
    }
    return () => {
      document.body.style.overflow = "";
    };
  }, [isOpen]);

  if (!isOpen) return null;

  const hasHeader = !!(title || header || showCloseButton);

  return (
    <div
      className="fixed inset-0 z-[100] flex items-end"
      data-testid={dataTestId}
    >
      <div
        className="absolute inset-0 bg-black/40"
        onClick={onClose}
        data-testid={`${dataTestId}-overlay`}
      />

      <div className={`relative w-full bg-white rounded-t-2xl ${heightClasses[heightMode]} flex flex-col shadow-xl animate-in slide-in-from-bottom duration-300 ${className}`}>
        {hasHeader && (
          <div className="flex items-center justify-between px-4 py-3 border-b border-border shrink-0">
            {header ? (
              <div className="flex-1 min-w-0">{header}</div>
            ) : (
              <div className="flex flex-col">
                {title && (
                  <h2
                    className="text-base font-bold text-foreground"
                    data-testid={`${dataTestId}-title`}
                  >
                    {title}
                  </h2>
                )}
                {subtitle && (
                  <p className="text-xs text-muted-foreground mt-0.5">{subtitle}</p>
                )}
              </div>
            )}
            {showCloseButton && (
              <button
                onClick={onClose}
                data-testid={`${dataTestId}-close`}
                className="p-1.5 hover:bg-muted rounded-full transition-colors ml-2"
              >
                <X className="h-5 w-5 text-foreground" />
              </button>
            )}
          </div>
        )}

        <div className="flex-1 overflow-y-auto min-h-0">{children}</div>

        {footer && (
          <div className="px-4 py-3 border-t border-border shrink-0">
            {footer}
          </div>
        )}
      </div>
    </div>
  );
}
